-- Hàm cập nhật danh sách người chơi
local function UpdateList()
    for _, v in pairs(Scroll:GetChildren()) do if v:IsA("TextButton") then v:Destroy() end end
    local yPos = 0
    for _, p in pairs(game.Players:GetPlayers()) do
        local btn = Instance.new("TextButton", Scroll)
        btn.Size = UDim2.new(1, 0, 0, 25)
        btn.Position = UDim2.new(0, 0, 0, yPos)
        btn.Text = p.Name
        btn.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
        btn.TextColor3 = Color3.new(1, 1, 1)
        btn.BorderSizePixel = 0
        btn.MouseButton1Click:Connect(function()
            selectedPlayer = p
            SelectedLabel.Text = "Đang chọn: " .. p.Name
        end)
        yPos = yPos + 27
    end
    Scroll.CanvasSize = UDim2.new(0, 0, 0, yPos)
end

-- Logic hiệu ứng Rainbow
AuraBtn.MouseButton1Click:Connect(function()
    if not selectedPlayer then return end
    auraActive = not auraActive
    
    if auraActive then
        AuraBtn.Text = "HỦY BỎ AURA"
        AuraBtn.BackgroundColor3 = Color3.fromRGB(120, 0, 0)
        
        currentHighlight = Instance.new("Highlight")
        currentHighlight.Parent = selectedPlayer.Character
        currentHighlight.FillTransparency = 0.5
        
        currentAura = Instance.new("Part")
        currentAura.Parent = workspace
        currentAura.Shape = Enum.PartType.Cylinder
        currentAura.Size = Vector3.new(0.4, 7, 7)
        currentAura.Anchored = true
        currentAura.CanCollide = false
        currentAura.Material = Enum.Material.Neon
        currentAura.Transparency = 0.4
        
        task.spawn(function()
            while auraActive and selectedPlayer and selectedPlayer.Character do
                local color = Color3.fromHSV(tick() % 5 / 5, 1, 1)
                if currentHighlight then 
                    currentHighlight.FillColor = color
                    currentHighlight.OutlineColor = color
                end
                if currentAura and selectedPlayer.Character:FindFirstChild("HumanoidRootPart") then
                    currentAura.Color = color
                    currentAura.CFrame = selectedPlayer.Character.HumanoidRootPart.CFrame * CFrame.Angles(0, 0, math.rad(90)) * CFrame.new(-3.2, 0, 0)
                end
                task.wait()
            end
            if currentHighlight then currentHighlight:Destroy() end
            if currentAura then currentAura:Destroy() end
        end)
    else
        AuraBtn.Text = "KÍCH HOẠT AURA"
        AuraBtn.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
        auraActive = false
    end
end)

UpdateList()
game.Players.PlayerAdded:Connect(UpdateList)
game.Players.PlayerRemoving:Connect(UpdateList)
