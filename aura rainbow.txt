local AuraBtn = Instance.new("TextButton", Main)

-- HỆ THỐNG LƯU TRỮ ĐỘC LẬP
local selectedPlayer = nil
local activeAuras = {} -- Lưu theo UserId: {Enabled = bool, Highlight = obj, Part = obj}

-- Thiết kế giao diện
Main.Size = UDim2.new(0, 220, 0, 320)

AuraBtn.Size = UDim2.new(0.8, 0, 0, 40)
AuraBtn.Position = UDim2.new(0.1, 0, 0, 260)
AuraBtn.Text = "KÍCH HOẠT"
AuraBtn.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
AuraBtn.TextColor3 = Color3.new(1, 1, 1)

-- Hàm cập nhật màu nút theo từng người
local function UpdateButtonUI()
    if selectedPlayer and activeAuras[selectedPlayer.UserId] then
        AuraBtn.Text = "TẮT AURA CHO NGƯỜI NÀY"
        AuraBtn.BackgroundColor3 = Color3.fromRGB(150, 0, 0)
    else
        AuraBtn.Text = "BẬT AURA CHO NGƯỜI NÀY"
        AuraBtn.BackgroundColor3 = Color3.fromRGB(0, 150, 0)
    end
end

-- Cập nhật danh sách
local function UpdateList()
    for _, v in pairs(Scroll:GetChildren()) do if v:IsA("TextButton") then v:Destroy() end end
    local yPos = 0
    for _, p in pairs(game.Players:GetPlayers()) do
        local btn = Instance.new("TextButton", Scroll)
        btn.Size = UDim2.new(1, 0, 0, 25)
        btn.Position = UDim2.new(0, 0, 0, yPos)
        btn.Text = p.Name
        btn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
        btn.TextColor3 = Color3.new(1, 1, 1)
        btn.MouseButton1Click:Connect(function()
            selectedPlayer = p
            SelectedLabel.Text = "Chọn: " .. p.Name
            UpdateButtonUI()
        end)
        yPos = yPos + 27
    end
    Scroll.CanvasSize = UDim2.new(0, 0, 0, yPos)
end

-- Logic Aura Độc Lập
AuraBtn.MouseButton1Click:Connect(function()
    if not selectedPlayer or not selectedPlayer.Character then return end
    local id = selectedPlayer.UserId
    local char = selectedPlayer.Character

    if activeAuras[id] then
        -- Tắt Aura của riêng người này
        activeAuras[id].Enabled = false
        activeAuras[id] = nil
    else
        -- Bật Aura mới cho người này
        local data = {Enabled = true}
        activeAuras[id] = data
        
        local hi = Instance.new("Highlight", char)
        local pt = Instance.new("Part", workspace)
        pt.Shape = Enum.PartType.Cylinder
        pt.Size = Vector3.new(0.3, 6, 6)
        pt.Anchored = true
        pt.CanCollide = false
        pt.Material = Enum.Material.Neon
        pt.Transparency = 0.5

        task.spawn(function()
            while data.Enabled and selectedPlayer and char.Parent do
                local c = Color3.fromHSV(tick() % 5 / 5, 1, 1)
                hi.FillColor = c
                hi.OutlineColor = c
                if char:FindFirstChild("HumanoidRootPart") then
                    pt.Color = c
                    pt.CFrame = char.HumanoidRootPart.CFrame * CFrame.Angles(0, 0, math.rad(90)) * CFrame.new(-3, 0, 0)
                end
                task.wait()
            end
            hi:Destroy()
            pt:Destroy()
        end)
    end
    UpdateButtonUI()
end)

UpdateList()
game.Players.PlayerAdded:Connect(UpdateList)
game.Players.PlayerRemoving:Connect(function(p) 
    if activeAuras[p.UserId] then activeAuras[p.UserId].Enabled = false end
    UpdateList() 
end)
